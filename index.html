tteBtn.onclick = async () => { if (tteFiles.length === 0) { return showHint('Harap pilih minimal 1 file (PDF atau Gambar) terlebih dahulu.'); } setBusy(true, 'Menggabungkan & Mengonversi File...'); try { const { PDFDocument, StandardFonts, rgb } = PDFLib; // 1) Gabungkan file sesuai urutan kustom pada tteFiles const outDoc = await PDFDocument.create(); for (const f of tteFiles) { if (f.type === 'image/jpeg' || f.name.toLowerCase().endsWith('.jpg') || f.name.toLowerCase().endsWith('.jpeg')) { const img = await outDoc.embedJpg(f.bytes); const page = outDoc.addPage([595.28, 841.89]); const imgDims = img.scaleToFit(page.getWidth() - 40, page.getHeight() - 40); page.drawImage(img, { x: (page.getWidth() - imgDims.width) / 2, y: (page.getHeight() - imgDims.height) / 2, width: imgDims.width, height: imgDims.height, }); } else if (f.type === 'image/png' || f.name.toLowerCase().endsWith('.png')) { const img = await outDoc.embedPng(f.bytes); const page = outDoc.addPage([595.28, 841.89]); const imgDims = img.scaleToFit(page.getWidth() - 40, page.getHeight() - 40); page.drawImage(img, { x: (page.getWidth() - imgDims.width) / 2, y: (page.getHeight() - imgDims.height) / 2, width: imgDims.width, height: imgDims.height, }); } else { const src = await PDFDocument.load(f.bytes.slice(0)); const copied = await outDoc.copyPages(src, src.getPageIndices()); copied.forEach(p => outDoc.addPage(p)); } } const names = [tteNama1.value, tteNama2.value] .map(s => (s || '').trim()) .filter(Boolean); if (names.length > 0) { const mergedBytes = await outDoc.save(); const pdfjsDoc = await pdfjsLib.getDocument({ data: mergedBytes.slice(0) }).promise; const fontReg = await outDoc.embedFont(StandardFonts.Helvetica); const labelSize = 9; const labelText = 'Ditandatangani secara elektronik'; const labelColor = rgb(0.55, 0.55, 0.55); const namesUpper = names.map(n => n.toUpperCase()); const outPages = outDoc.getPages(); let totalStamped = 0; for (let i = 1; i <= pdfjsDoc.numPages; i++) { setBusy(true, `Mencari nama di halaman ${i}/${pdfjsDoc.numPages}...`); const page = await pdfjsDoc.getPage(i); const tc = await page.getTextContent(); // Perbaikan: Perluas toleransi klaster baris agar teks berdekatan terbaca sebagai satu kesatuan const rows = clusterRows(tc.items, 6); const outPage = outPages[i - 1]; if (!outPage) continue; namesUpper.forEach(nameUpper => { let lowestMatch = null; rows.forEach(row => { const lineText = row.map(it => it.str).join(' ').replace(/\s+/g, ' ').trim().toUpperCase(); // Cek apakah baris mengandung nama yang dicari if (!lineText || !lineText.includes(nameUpper)) return; const startItem = findNameStartItemInRow(row, nameUpper) || row[0]; const x = startItem.transform[4]; const y = startItem.transform[5]; const lineHeight = startItem.height || 10; // Ambil posisi vertikal tertinggi/terakhir di halaman jika nama muncul if (lowestMatch === null || y < lowestMatch.y) { lowestMatch = { x, y, lineHeight }; } }); if (lowestMatch) { outPage.drawText(labelText, { x: lowestMatch.x, y: lowestMatch.y + lowestMatch.lineHeight + 4, // Sedikit dinaikkan jaraknya agar pas di atas nama size: labelSize, font: fontReg, color: labelColor }); totalStamped++; } }); } if (totalStamped === 0) { showHint('Nama tidak ditemukan di teks dokumen. Pastikan ejaan sama persis.'); } else { showHint(`Berhasil menempel ${totalStamped} tanda tangan.`); } } setBusy(true, 'Menyiapkan & Mengompres PDF...'); const compVal = compressionLevel.value; let saveOptions = { useObjectStreams: true }; if (compVal === 'medium' || compVal === 'high') { saveOptions.addDefaultPage = false; } const bytes = await outDoc.save(saveOptions); downloadBlob(bytes, 'application/pdf', 'Dokumen_Gabungan_TTE.pdf'); } catch (err) { console.error(err); showHint('Terjadi kesalahan saat memproses file.'); } setBusy(false); };
